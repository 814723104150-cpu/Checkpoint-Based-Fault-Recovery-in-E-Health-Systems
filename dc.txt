import time
import random
import pickle

# Simulated patient data
patients = {
    "P001": {"heart_rate": 72, "bp": "120/80"},
    "P002": {"heart_rate": 85, "bp": "130/90"},
    "P003": {"heart_rate": 68, "bp": "110/70"},
}

checkpoint_file = "checkpoint.pkl"

def save_checkpoint(data):
    with open(checkpoint_file, "wb") as f:
        pickle.dump(data, f)
    print("[Checkpoint] System state saved successfully.\n")

def load_checkpoint():
    with open(checkpoint_file, "rb") as f:
        data = pickle.load(f)
    print("[Recovery] System recovered from last checkpoint.\n")
    return data

def simulate_system():
    global patients
    iteration = 0
    save_checkpoint(patients)  # Initial checkpoint

    while iteration < 10:
        iteration += 1
        print(f"--- Iteration {iteration} ---")

        # Randomly update some patient data
        for pid in patients:
            patients[pid]["heart_rate"] += random.randint(-3, 3)

        print("Current patient data:", patients)

        # Random fault simulation
        if random.random() < 0.25:
            print("\n[Error] System crash detected!")
            patients = load_checkpoint()  # Recovery
        else:
            save_checkpoint(patients)

        time.sleep(1)

    print("\n[System] Monitoring completed successfully.")

if __name__ == "__main__":
    simulate_system()
